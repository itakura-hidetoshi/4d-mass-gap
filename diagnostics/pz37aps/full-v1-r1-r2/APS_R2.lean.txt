    (H : ℕ) (gamma : PeriodicHypercubicEvenVertex H → G)

      SG H 2 (fun v => gamma (At H 1 v)) ((E H 2 (b, x)).1 (j1 H)) := by
  funext e
  change (E H 2 (BG H 2 gamma b, OG H 2 gamma x)).1 (j1 H) e =
    gamma (At H 1 e.1) * (E H 2 (b, x)).1 (j1 H) e *
      (gamma (At H 1 (periodicHypercubicEvenSpatialSliceShift H e.1 e.2)))⁻¹
  rw [periodicHypercubicEvenSpatialSliceVertexAtTime_shift]
  let edge := periodicHypercubicEvenSpatialSliceLinkAtTime H 1 e
  change _ = gamma (periodicHypercubicEdgeSource (PeriodicHypercubicEvenSideLength H) edge) *
    (E H 2 (b, x)).1 (j1 H) e *
      (gamma (periodicHypercubicEdgeTarget (PeriodicHypercubicEvenSideLength H) edge))⁻¹
  by_cases hH : H = 0
  · let ef := (periodicHypercubicEvenFixedEdgeEquivTwoSpatialSlices H).symm (Sum.inr e)
    have hj : j1 H = ⟨H + 1, by
        simp [periodicHypercubicEvenPositiveHalfCylinderSlabCount]⟩ := by
      apply Fin.ext
      simp [p4Q2PZ37AMX_positiveTimeOne, hH]
    have hedge : edge = ef.1 := by
      dsimp [edge, ef, periodicHypercubicEvenFixedEdgeEquivTwoSpatialSlices,
        periodicHypercubicEvenSpatialSliceSumToFixedEdge,
        periodicHypercubicEvenAntipodalSpatialSliceLinkToFixedEdge,
        periodicHypercubicEvenPrimaryAntipodalSpatialSliceLinkEquiv,
        periodicHypercubicEvenPrimaryAntipodalSpatialSliceVertexEquiv,
        periodicHypercubicEvenPrimaryToAntipodalSpatialSliceVertex]
      unfold periodicHypercubicEvenSpatialSliceLinkAtTime
      have ht : (1 : ZMod (PeriodicHypercubicEvenSideLength H)) =
          (((H + 1 : ℕ) : ZMod (PeriodicHypercubicEvenSideLength H))) := by simp [hH]
      rw [ht, periodicHypercubicEvenSpatialSliceVertexAtTime_halfPeriod_eq]
    simp only [hj, periodicHypercubicEvenSpecialUnitaryPositiveHalfClosureTransfer_antipodal_apply, hedge]
    rfl
  · let k : Fin H := ⟨0, Nat.pos_of_ne_zero hH⟩

    periodicHypercubicEvenPositiveOpenHalfGaugeTransform_measurePreserving H 2 gamma

  unfold p4Q2PZ37ANF_originalTimeOneWilsonBoundaryGramMoment
    periodicHypercubicEvenBoundaryObservableGramFeature

    Gram H 2 (by norm_num) beta hb b x * F H c0 c1 (b, x) ∂mu

    _ = ∫ x, f (T x) ∂mu := (hmp.integral_comp hme f).symm
