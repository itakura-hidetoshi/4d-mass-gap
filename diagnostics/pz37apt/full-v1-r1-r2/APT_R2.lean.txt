    (H : ℕ) [Nontrivial G] (beta : ℝ) (hb : 0 ≤ beta) :
    ‖A H beta hb‖ = ‖C H beta hb‖ := by

  · apply ContinuousLinearMap.opNorm_le_bound _ (norm_nonneg _)

    let p : P H 2 := ⟨GP H 2 f,
      periodicHypercubicEvenSpecialUnitarySpatialSliceGaussLawProjection_mem H 2 f⟩
    have hp : ‖p‖ ≤ ‖f‖ := by

      exact Submodule.norm_starProjection_apply_le
