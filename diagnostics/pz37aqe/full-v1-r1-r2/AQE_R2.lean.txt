/-- A finite real midpoint recurrence implies a dyadic power bound,
including the zero-base case, without division or an infinite-time premise. -/
theorem p4Q2PZ37AQE_finiteDyadicPower
    (b : ℝ) (a : ℕ → ℝ) (ha : ∀ j, 0 ≤ a j)
    (n : ℕ) (hrec : ∀ j < n, a j ^ 2 ≤ b * a (j + 1)) :
    a 0 ^ ((2 : ℕ) ^ n) ≤ b ^ ((2 : ℕ) ^ n - 1) * a n := by
  induction n with
  | zero => simp
  | succ n ih =>
      have hprev := ih (fun j hj => hrec j (Nat.lt_trans hj (Nat.lt_succ_self n)))
      have hsq : (a 0 ^ ((2 : ℕ) ^ n)) ^ 2 ≤
          (b ^ ((2 : ℕ) ^ n - 1) * a n) ^ 2 := by
        simpa only [pow_two] using
          mul_self_le_mul_self (pow_nonneg (ha 0) ((2 : ℕ) ^ n)) hprev
      have hexp : (2 : ℕ) ^ (n + 1) - 1 = ((2 : ℕ) ^ n - 1) * 2 + 1 := by
        have hp : 0 < (2 : ℕ) ^ n := pow_pos (by norm_num) _
        rw [pow_succ]
        omega
      calc
        a 0 ^ ((2 : ℕ) ^ (n + 1)) = (a 0 ^ ((2 : ℕ) ^ n)) ^ 2 := by
          rw [pow_succ (2 : ℕ) n, pow_mul]
        _ ≤ _ := hsq
        _ = (b ^ ((2 : ℕ) ^ n - 1)) ^ 2 * (a n) ^ 2 := by ring
        _ ≤ (b ^ ((2 : ℕ) ^ n - 1)) ^ 2 * (b * a (n + 1)) :=
          mul_le_mul_of_nonneg_left (hrec n (Nat.lt_succ_self n)) (sq_nonneg _)
        _ = b ^ ((2 : ℕ) ^ (n + 1) - 1) * a (n + 1) := by
          rw [hexp, pow_add, pow_mul, pow_one]
          ring

/-- Finite dyadic iteration on the genuine Wilson synthesis image.
All intermediate times are justified by the single terminal support budget. -/

    (beta : ℝ) (hb : 0 ≤ beta)

    ‖Read H beta hb (Local H k u)‖ ^ ((2 : ℕ) ^ n) ≤
      ‖Read H beta hb u‖ ^ ((2 : ℕ) ^ n - 1) *
        ‖Read H beta hb (Local H ((2 : ℕ) ^ n * k) u)‖ := by
  let a : ℕ → ℝ := fun j => ‖Read H beta hb (Local H ((2 : ℕ) ^ j * k) u)‖
  have hrec (j : ℕ) (hj : j < n) : a j ^ 2 ≤ ‖Read H beta hb u‖ * a (j + 1) := by
    have htime : (2 : ℕ) ^ j * k + (2 : ℕ) ^ j * k = (2 : ℕ) ^ (j + 1) * k := by
      rw [pow_succ]
      ring
    have hpow : (2 : ℕ) ^ (j + 1) ≤ (2 : ℕ) ^ n :=
      pow_le_pow_right' (by norm_num : (1 : ℕ) ≤ 2) (Nat.succ_le_of_lt hj)
    have hmul := Nat.mul_le_mul_right k hpow
    have hkk : ∀ e ∈ s, (e.1 0).val + (2 : ℕ) ^ j * k + (2 : ℕ) ^ j * k ≤ H := by
      intro e he
      have h := hbudget e he
      calc
        _ = (e.1 0).val + ((2 : ℕ) ^ j * k + (2 : ℕ) ^ j * k) := by omega
        _ = (e.1 0).val + (2 : ℕ) ^ (j + 1) * k := by rw [htime]
        _ ≤ (e.1 0).val + (2 : ℕ) ^ n * k := Nat.add_le_add_left hmul _
        _ ≤ H := h
    have hmid := p4Q2PZ37AQE_localMidpointNorm H ((2 : ℕ) ^ j * k) s beta hb u hu hp hkk
    rw [htime] at hmid
    exact hmid
  have ha : ∀ j, 0 ≤ a j := fun _ => norm_nonneg _
  have h := p4Q2PZ37AQE_finiteDyadicPower ‖Read H beta hb u‖ a ha n hrec
  simpa only [a, pow_zero, one_mul] using h
