/-- The existing physical transfer coefficient is precisely the pairing of
the actual boundary vacuum with the actual endpoint tensor. -/

    (H : ℕ) (beta : ℝ) (hb : 0 ≤ beta) (f g : Phys H 2) :
    inner ℝ (Probe H f g) (Vac H 2 (by norm_num) beta hb) =
      NormZ H 2 (by norm_num) beta hb *
        inner ℝ (HalfTr H 2 (by norm_num) beta hb f) g := by
  let fl : Lp ℝ 2 (Mu H 2) := f
  let gl : Lp ℝ 2 (Mu H 2) := g
  let F : Closure H G → ℝ := fun z =>
    Gram H 2 (by norm_num) beta hb z.1 z.2 * fl ((E H 2 z).1 0) *
      gl ((E H 2 z).1 (Fin.last (Slab H)))
  have hF : Integrable F (CM H 2) :=
    periodicHypercubicEvenBoundaryCompletedPositiveGramFeature_gaussEndpoints_integrable
      H 2 (by norm_num) beta hb f g
  have hTransfer : (∫ z, F z ∂(CM H 2)) =
      NormZ H 2 (by norm_num) beta hb *
        inner ℝ (HalfTr H 2 (by norm_num) beta hb f) g :=
    periodicHypercubicEvenBoundaryCompletedPositiveGramFeature_gaussEndpoint_integral_eq_normalizedPhysicalTransfer
      H 2 (by norm_num) beta hb f g
  have hFubini : (∫ z, F z ∂(CM H 2)) =
      ∫ b : B H 2, (fl (PairE H G b).1 * gl (PairE H G b).2) *
        VacM H 2 (by norm_num) beta hb b ∂(BM H 2) := by
    calc
      _ = ∫ b : B H 2, (∫ x : O H 2, F (b,x) ∂(OM H 2)) ∂(BM H 2) :=
        MeasureTheory.integral_prod F hF
      _ = _ := by
        apply integral_congr_ae
        filter_upwards with b
        let k := fl (PairE H G b).1 * gl (PairE H G b).2
        calc
          _ = ∫ x : O H 2, k * Gram H 2 (by norm_num) beta hb b x ∂(OM H 2) := by
            apply integral_congr_ae
            filter_upwards with x
            dsimp [F, k]
            rw [p4Q2PZ37AQW_left_eq H b x, p4Q2PZ37AQW_right_eq H b x]
            ring
          _ = k * VacM H 2 (by norm_num) beta hb b := by
            rw [integral_const_mul] <;> rfl
  rw [← hTransfer, hFubini, MeasureTheory.L2.inner_def]
  apply integral_congr_ae
  filter_upwards [p4Q2PZ37AQO_endpointProbe_coeFn H f g,
    periodicHypercubicEvenBoundaryVacuumL2_coeFn H 2 (by norm_num) beta hb] with b hf hv

theorem p4Q2PZ37AQW_partitionFactor_eq_overlap
    (H : ℕ) (beta : ℝ) (hb : 0 ≤ beta) :
    PF H beta hb = inner ℝ (TopProbe H beta hb) (Vac H 2 (by norm_num) beta hb) := by
  let t := Tr H 2 (by norm_num) beta hb
  let o := Omega H 2 (by norm_num) beta hb
  have he : t o = ‖t‖ • o :=
    periodicHypercubicEvenSpecialUnitaryPhysicalOneSlabNonnegativeTopEigenvector_eigen
      H 2 (by norm_num) beta hb
  have hPow : HalfTr H 2 (by norm_num) beta hb o = ‖t‖^(H+1) • o :=
    p4Q2PZ37AQW_pow_eigen t o ‖t‖ he (H+1)
  have ho : ‖o‖ = 1 :=
    periodicHypercubicEvenSpecialUnitaryPhysicalOneSlabNonnegativeTopEigenvector_norm
      H 2 (by norm_num) beta hb
  have hInner : inner ℝ (HalfTr H 2 (by norm_num) beta hb o) o = ‖t‖^(H+1) := by
    calc
      _ = inner ℝ (‖t‖^(H+1) • o) o :=

/-- A nonzero uniform partition floor is equivalent to a uniform strict
upper bound on the orthogonal residual of the same boundary vacuum.
This criterion is not a proof that either side holds. -/
theorem p4Q2PZ37AQW_partitionFloor_iff_residualMargin
    (beta : ℝ) (hb : 0 ≤ beta) :
    (∃ c : ℝ, 0 < c ∧ ∀ H : ℕ, c ≤ PF H beta hb) ↔
    (∃ e : ℝ, 0 < e ∧ ∀ H : ℕ,
      ‖Vac H 2 (by norm_num) beta hb - PF H beta hb • TopProbe H beta hb‖^2 ≤ 1-e) := by
  constructor
  · rintro ⟨c,hc,h⟩
    refine ⟨c^2,by positivity,?_⟩
    intro H
    rw [p4Q2PZ37AQW_vacuum_residual_normSq]
    have hsq := pow_le_pow_left₀ hc.le (h H) 2
    linarith only [hsq]
  · rintro ⟨e,he,h⟩
    refine ⟨Real.sqrt e,Real.sqrt_pos.2 he,?_⟩
    intro H
    have hr := h H
    rw [p4Q2PZ37AQW_vacuum_residual_normSq] at hr
    have hp := (p4Q2PZ37AQW_partitionFactor_pos_le_one H beta hb).1
    have hsq : e ≤ (PF H beta hb)^2 := by linarith only [hr]
    have h' := Real.sqrt_le_sqrt hsq
    rw [Real.sqrt_sq hp.le] at h'
    exact h'
