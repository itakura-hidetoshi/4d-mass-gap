      f (path 0) * UK H 2 beta path U * a (path j) *
        g (path (Fin.last (Slab H))) ∂(PathMu H 2)) =
    ∫ path : Path H 2,
      f (path 0) * TK H 2 beta path * a (path j) *
        g (path (Fin.last (Slab H))) ∂(PathMu H 2) := by
  let e := periodicHypercubicEvenSpecialUnitaryPositiveHalfTemporalGaugeSpatialPathMeasurableEquiv
    H 2 U
  let F : Path H 2 → ℝ := fun path =>
    f (path 0) * TK H 2 beta path * a (path j) *
      g (path (Fin.last (Slab H)))
  have hTerminal :=
    periodicHypercubicEvenSpecialUnitaryPositiveHalfTerminalGaugeInvariant_ae H 2 g hg U
  calc
    _ = ∫ path : Path H 2, F (e path) ∂(PathMu H 2) := by
      apply integral_congr_ae
      filter_upwards [hTerminal] with path ht
      have he : e path = Gauge H 2 U path := rfl
      have hZero : (e path) 0 = path 0 := by
        rw [he]
        have h0 : (0 : Fin (Slab H + 1)) = ⟨0, Nat.zero_lt_succ _⟩ := by
          apply Fin.ext
          rfl
        rw [h0]
        exact periodicHypercubicEvenSpecialUnitaryPositiveHalfCylinderTemporalGaugeSpatialPath_zero
          H 2 U path
      have hObs : a ((e path) j) = a (path j) := by
        rw [he]
        exact ha
          (periodicHypercubicEvenSpecialUnitaryPositiveHalfCylinderCumulativeGauge H 2 U j)
          (path j)
      have hLast : g ((e path) (Fin.last (Slab H))) =
          g (path (Fin.last (Slab H))) := by
        rw [he]
        exact ht
      have hK : UK H 2 beta path U = TK H 2 beta (e path) := by
        rw [he]
        exact periodicHypercubicEvenSpecialUnitaryPositiveHalfCylinderUnfixedPathKernel_eq_temporalGauge
          H 2 beta path U

            a ((e path) j) * g ((e path) (Fin.last (Slab H))) := by
          rw [hZero, hObs, hLast, ← hK]
        _ = F (e path) := rfl

/-- Transport a normalized, endpoint-tested completed OS feature with one
literal bounded gauge-invariant slice insertion to the actual temporal-gauge
path amplitude. Integrability is inherited from the exact endpoint theorem,
then multiplied by the bounded inserted observable. -/
theorem p4Q2PZ37APM_originalClosureInsertedIntegral_eq_nativeAmplitude
    (H : ℕ) [Nontrivial G] (beta : ℝ) (hb : 0 ≤ beta)
    (j : Fin (Slab H + 1)) (a : Aobs H 2)
    (ha : periodicHypercubicEvenSpecialUnitarySpatialSliceObservableGaugeInvariant H 2 a)
    (f g : Phys H 2) :
    (∫ z : Closure H G,
      Gram H 2 (by norm_num) beta hb z.1 z.2 *
        (f : Lp ℝ 2 (Mu H 2)) ((E H 2 z).1 0) *
        (g : Lp ℝ 2 (Mu H 2)) ((E H 2 z).1 (Fin.last (Slab H))) *
        a ((E H 2 z).1 j) ∂(ClosureMu H 2)) =
    NormZ H 2 (by norm_num) beta hb *
      NativeAmp H 2 beta (Slab H) j a
        (f : Lp ℝ 2 (Mu H 2)) (g : Lp ℝ 2 (Mu H 2)) := by
  let e := E H 2
  let c := NormZ H 2 (by norm_num) beta hb
  let fl : Lp ℝ 2 (Mu H 2) := f
  let gl : Lp ℝ 2 (Mu H 2) := g
  let F0 : Path H 2 × Field H 2 → ℝ := fun p =>
    c * (fl (p.1 0) * UK H 2 beta p.1 p.2 *
      gl (p.1 (Fin.last (Slab H))))
  let F : Path H 2 × Field H 2 → ℝ := fun p => F0 p * a (p.1 j)
  have hF0 : Integrable F0 (NestedMu H 2) :=
    periodicHypercubicEvenSpecialUnitaryPositiveHalf_normalizedGaussEndpointKernel_integrable
      H 2 (by norm_num) beta hb f g
  have hMeas : Measurable (fun p : Path H 2 × Field H 2 => a (p.1 j)) :=
    a.continuous.measurable.comp ((measurable_pi_apply j).comp measurable_fst)
  have hBound : ∀ᵐ p : Path H 2 × Field H 2 ∂(NestedMu H 2),
      ‖a (p.1 j)‖ ≤ ‖a‖ :=
    Filter.Eventually.of_forall (fun p => a.norm_coe_le_norm (p.1 j))
  have hF : Integrable F (NestedMu H 2) :=
    hF0.mul_bdd hMeas.aestronglyMeasurable hBound
  have hMP : MeasurePreserving e (ClosureMu H 2) (NestedMu H 2) :=
    periodicHypercubicEvenSpecialUnitaryPositiveHalfClosureTransferMeasurableEquiv_measurePreserving_explicitHaar
      H 2
  calc
    _ = ∫ z : Closure H G, F (e z) ∂(ClosureMu H 2) := by
      apply integral_congr_ae
      filter_upwards with z
      dsimp [F, F0, c, e, fl, gl]
      rw [periodicHypercubicEvenBoundaryCompletedPositiveGramFeature_eq_normalizedUnfixedPathKernel
        H 2 (by norm_num) beta hb z.1 z.2]
      ring
    _ = ∫ p, F p ∂(NestedMu H 2) := by
      exact hMP.integral_comp' F
    _ = ∫ U : Field H 2, (∫ path : Path H 2, F (path, U) ∂(PathMu H 2))
        ∂(FieldMu H 2) := by
      simpa only [periodicHypercubicEvenSpecialUnitaryPositiveHalfExplicitNestedHaarMeasure] using
        (MeasureTheory.integral_prod_symm F hF)
    _ = c * (∫ U : Field H 2,
        (∫ path : Path H 2,
          fl (path 0) * UK H 2 beta path U * a (path j) *
            gl (path (Fin.last (Slab H))) ∂(PathMu H 2))
        ∂(FieldMu H 2)) := by
      rw [← integral_const_mul]
      apply integral_congr_ae
      filter_upwards with U
      rw [← integral_const_mul]
      apply integral_congr_ae
      filter_upwards with path
      dsimp [F, F0]
      ring
    _ = c * (∫ path : Path H 2,
        fl (path 0) * TK H 2 beta path * a (path j) *
          gl (path (Fin.last (Slab H))) ∂(PathMu H 2)) := by
      exact congrArg (fun y : ℝ => c * y)
        (p4Q2PZ37APM_originalGaussSliceIteratedHaarIntegral_eq_temporalGauge
          H beta j a ha fl gl g.property)
    _ = _ := rfl
