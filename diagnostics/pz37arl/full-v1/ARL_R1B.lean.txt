private theorem p4Q2PZ37ARL_primitive (p : ℕ) (x : ℝ) :
    (∫ t in 0..x, ((p : ℝ)+1)*t^p) = x^(p+1) := by
  rw [intervalIntegral.integral_const_mul, intervalIntegral.integral_pow,
    zero_pow (Nat.succ_ne_zero p), sub_zero]
  have hp : (p : ℝ)+1 ≠ 0 := by positivity
  exact mul_div_cancel_left₀ _ hp

private theorem p4Q2PZ37ARL_layer_aux {ι : Type i} (rho : ι → ℝ)
    (p : ℕ) (a : ℝ) (h0 : ∀ k, 0 ≤ rho k)
    (ha : HasSum (fun k => (rho k)^(p+1)) a) :
    ENNReal.ofReal a = ∫⁻ t in Ioi (0:ℝ), N rho t *
      ENNReal.ofReal (((p : ℝ)+1)*t^p) := by
  letI : MeasurableSpace ι := ⊤
  letI : MeasurableSingletonClass ι := ⟨fun _ => trivial⟩
  have hm : Measurable rho := by
    intro s _
    trivial
  have hg : Continuous (fun t : ℝ => ((p : ℝ)+1)*t^p) := by fun_prop
  have hl := lintegral_comp_eq_lintegral_meas_le_mul
    (f := rho) (g := fun t : ℝ => ((p : ℝ)+1)*t^p)
    (Measure.count : Measure ι) (Eventually.of_forall h0) hm.aemeasurable
    (fun t _ => hg.intervalIntegrable 0 t)
    (by
      filter_upwards [self_mem_ae_restrict (measurableSet_Ioi : MeasurableSet (Ioi (0:ℝ)))]
        with t ht
      exact mul_nonneg (by positivity) (pow_nonneg ht.le _))
  have hprim (k : ι) :
      (∫ t in 0..rho k, ((p : ℝ)+1)*t^p) = (rho k)^(p+1) :=
    p4Q2PZ37ARL_primitive p (rho k)
  simp_rw [hprim] at hl
  have hleft : (∫⁻ k, ENNReal.ofReal ((rho k)^(p+1)) ∂(Measure.count : Measure ι)) =
      ENNReal.ofReal a := by
    calc
      _ = ∑' k, ENNReal.ofReal ((rho k)^(p+1)) := lintegral_count _
      _ = ENNReal.ofReal (∑' k, (rho k)^(p+1)) :=
        (ENNReal.ofReal_tsum_of_nonneg (fun k => pow_nonneg (h0 k) _) ha.summable).symm
      _ = ENNReal.ofReal a := congrArg ENNReal.ofReal ha.tsum_eq
  rw [hleft] at hl
  refine hl.trans ?_
  apply lintegral_congr
  intro t
  simp only [p4Q2PZ37ARL_spectralCount,
    Measure.count_apply (show MeasurableSet {k : ι | t ≤ rho k} by trivial)]

private theorem p4Q2PZ37ARL_trunc_aux {ι : Type i} (rho : ι → ℝ)
    (g : ℝ → ℝ≥0∞) (hcap : ∀ k, rho k ≤ 1) :
    (∫⁻ t in Ioi (0:ℝ), N rho t * g t) =
      ∫⁻ t in Ioc (0:ℝ) 1, N rho t * g t := by
  let f : ℝ → ℝ≥0∞ := fun t => N rho t * g t
  have hs := lintegral_inter_add_diff (μ := volume) f (Ioi (0:ℝ))
    (measurableSet_Ioc : MeasurableSet (Ioc (0:ℝ) 1))
  have hi : Ioi (0:ℝ) ∩ Ioc 0 1 = Ioc 0 1 := by
    ext t
    simp only [mem_inter_iff, mem_Ioi, mem_Ioc]
    tauto
  have hz : (∫⁻ t in Ioi (0:ℝ) \ Ioc 0 1, f t) = 0 := by
    apply setLIntegral_eq_zero (measurableSet_Ioi.diff measurableSet_Ioc)
    intro t ht
    have ht1 : 1 < t := by
      by_contra! h
      exact ht.2 ⟨ht.1,h⟩
    have he : {k : ι | t ≤ rho k} = ∅ := by
      apply Set.eq_empty_of_forall_notMem
      intro k hk
      exact (not_le_of_gt ht1) (hk.trans (hcap k))
    simp [f, p4Q2PZ37ARL_spectralCount, he]
  rw [hi, hz, add_zero] at hs
  exact hs.symm

/-- Exact layer-cake formula for the genuine complete moment at every
degree, on the full positive threshold axis. -/
theorem p4Q2PZ37ARL_moment_layercake_Ioi
    (H : ℕ) (beta : ℝ) (hb : 0 ≤ beta) (n : ℕ)
    {ι : Type i} (b : HilbertBasis ι ℝ (K H 2 h2 beta hb))
    (rho : ι → ℝ) (he : ∀ k, Step H 2 h2 beta hb (b k) = rho k • b k) :
    ENNReal.ofReal (M H beta hb n) =
      ∫⁻ t in Ioi (0:ℝ), N rho t * ENNReal.ofReal (2*((n:ℝ)+1)*t^(2*n+1)) := by
  have h0 (k : ι) : 0 ≤ rho k :=
    (p4Q2PZ37ARA_eigenvalue_bounds H beta hb (b k) (rho k)
      (b.orthonormal.norm_eq_one k) (he k)).1
  have hm : HasSum (fun k => (rho k)^((2*n+1)+1)) (M H beta hb n) := by
    rw [show 2*n+1+1 = 2*(n+1) by omega]
    exact p4Q2PZ37ARD_moment_hasSum H beta hb n b rho he
  have hw : ((2*n+1 : ℕ) : ℝ)+1 = 2*((n:ℝ)+1) := by
    push_cast <;> ring
  simpa only [hw] using p4Q2PZ37ARL_layer_aux rho (2*n+1) (M H beta hb n) h0 hm
