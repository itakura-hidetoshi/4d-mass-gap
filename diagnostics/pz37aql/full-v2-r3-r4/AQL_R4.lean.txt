/-- Adding a constant preserves the specified literal finite support. -/
theorem p4Q2PZ37AQL_supported_add_const
    (H : ℕ) (s : Set (PeriodicHypercubicEvenEdge H)) (u : Obs H) (t : ℝ)
    (hu : Supported H s (fun A => u ((P H).positiveRestriction A))) :
    Supported H s (fun A => (u + t • (1 : Obs H)) ((P H).positiveRestriction A)) := by
  intro A B hAB
  change u ((P H).positiveRestriction A) + t * 1 =
    u ((P H).positiveRestriction B) + t * 1
  have heq : u ((P H).positiveRestriction A) = u ((P H).positiveRestriction B) :=
    hu A B hAB
  exact congrArg (fun z : ℝ => z + t * 1) heq

    (H : ℕ) (beta : ℝ) (hb : 0 ≤ beta) (u : Obs H) (t : ℝ) :
    Mean H beta hb (u + t • (1 : Obs H)) = Mean H beta hb u + t := by

/-- On the genuine finite Wilson model, the raw AQK prefactor is unbounded
on a constant-shift fiber with nonzero initial connected correlation.
This is an obstruction to a representative-independent raw prefactor,
not a failure of actual relative contraction or centered coercivity. -/
theorem p4Q2PZ37AQL_rawPrefactor_unbounded_on_centered_fiber
    (H k : ℕ) (S : Finset (PeriodicHypercubicEvenEdge H))
    (beta : ℝ) (hb : 0 ≤ beta) (hThreshold : Rho beta < 1)
    (u : Obs H) (x : O H 2)
    (hu : Supported H (↑S : Set (PeriodicHypercubicEvenEdge H))
      (fun A => u ((P H).positiveRestriction A)))
    (hp : ∀ e ∈ S, periodicHypercubicEvenEdgeSide H e = ReflectionEdgeSide.positive)
    (hbudget : ∀ e ∈ S, (e.1 0).val ≤ H)
    (hS : 0 < S.card) (hbase : 0 < EvenC H k 0 beta hb u)
    (K : ℝ) :
    ∃ t : ℝ, let v := u + t • (1 : Obs H)
      Supported H (↑S : Set (PeriodicHypercubicEvenEdge H))
        (fun A => v ((P H).positiveRestriction A)) ∧
      Center H beta hb v = Center H beta hb u ∧
      EvenC H k 0 beta hb v = EvenC H k 0 beta hb u ∧
      0 < EvenC H k 0 beta hb v ∧
      K * EvenC H k 0 beta hb v < Bnd H S beta v := by
  let a := EvenC H k 0 beta hb u
  let d := 1 - Rho beta
  let M := |K * a| + d + 1
  let t := M - u x
  let v := u + t • (1 : Obs H)
  have hd : 0 < d := sub_pos.mpr hThreshold
  have hM : 0 < M := by
    dsimp [M]
    linarith [abs_nonneg (K * a)]
  have hvx : v x = M := by
    change u x + (M - u x) * 1 = M
    ring
  have hnorm : M ≤ ‖v‖ := by
    have hh := BoundedContinuousFunction.norm_coe_le_norm v x
    rw [hvx, Real.norm_eq_abs, abs_of_pos hM] at hh
    exact hh
  have hc : (1 : ℝ) ≤ (S.card : ℝ) := by
    exact_mod_cast (show 1 ≤ S.card by omega)
  have hnum : M ≤ (S.card : ℝ) * (2 * ‖v‖) := by
    calc
      _ ≤ ‖v‖ := hnorm
      _ ≤ 2 * ‖v‖ := by linarith [norm_nonneg v]
      _ = 1 * (2 * ‖v‖) := (one_mul _).symm
      _ ≤ _ := mul_le_mul_of_nonneg_right hc (by positivity)
  have hsq : M ^ 2 ≤ ((S.card : ℝ) * (2 * ‖v‖)) ^ 2 :=
    pow_le_pow_left₀ hM.le hnum 2
  have habs : |K * a| < M := by
    dsimp [M]
    linarith
  have hdM : d ≤ M := by
    dsimp [M]
    linarith [abs_nonneg (K * a)]
  have hlt : K * a * d < M ^ 2 := by
    calc
      _ ≤ |K * a| * d := mul_le_mul_of_nonneg_right (le_abs_self _) hd.le
      _ < M * d := mul_lt_mul_of_pos_right habs hd
      _ ≤ M * M := mul_le_mul_of_nonneg_left hdM hM.le
      _ = _ := (pow_two M).symm
  have hEq : EvenC H k 0 beta hb v = a := by
    exact p4Q2PZ37AQL_evenConnected_add_const H k 0 (↑S) beta hb u t hu hp
      (by simpa using hbudget)
  refine ⟨t, p4Q2PZ37AQL_supported_add_const H (↑S) u t hu,
    p4Q2PZ37AQL_center_add_const H beta hb u t, hEq, ?_, ?_⟩
  · rw [hEq]
    exact hbase
  · change K * EvenC H k 0 beta hb v < Bnd H S beta v
    rw [hEq]
    change K * a < ((S.card : ℝ) * (2 * ‖v‖)) ^ 2 / d
    exact (lt_div_iff₀ hd).2 (hlt.trans_le hsq)

/-- On a chosen representative with a norm-to-initial-OS bound, a bounded
support cardinality and denominator floor suffice for AQK's relative
amplitude budget. The norm-to-OS estimate is a hypothesis, not a result. -/
theorem p4Q2PZ37AQL_prefactor_le_of_normBudget
    (H k : ℕ) (S : Finset (PeriodicHypercubicEvenEdge H))
    (beta : ℝ) (hb : 0 ≤ beta) (u : Obs H)
    (N C d : ℝ) (hd : 0 < d) (hden : d ≤ 1 - Rho beta)
    (hcard : (S.card : ℝ) ≤ N)
    (hnorm : ‖u‖ ^ 2 ≤ C * EvenC H k 0 beta hb u) :
    Bnd H S beta u ≤ (4 * N ^ 2 * C / d) * EvenC H k 0 beta hb u := by
  have hcard2 : (S.card : ℝ) ^ 2 ≤ N ^ 2 :=
    pow_le_pow_left₀ (by positivity) hcard 2
  have hnum : ((S.card : ℝ) * (2 * ‖u‖)) ^ 2 ≤
      4 * N ^ 2 * (C * EvenC H k 0 beta hb u) := by
    calc
      _ = 4 * (S.card : ℝ) ^ 2 * ‖u‖ ^ 2 := by ring
      _ ≤ 4 * N ^ 2 * ‖u‖ ^ 2 :=
        mul_le_mul_of_nonneg_right (mul_le_mul_of_nonneg_left hcard2 (by norm_num)) (sq_nonneg _)
      _ ≤ _ := mul_le_mul_of_nonneg_left hnorm (by positivity)
  unfold p4Q2PZ37AQK_prefactor
  calc
    _ ≤ ((S.card : ℝ) * (2 * ‖u‖)) ^ 2 / d :=
      div_le_div_of_nonneg_left (sq_nonneg _) hd hden
    _ ≤ (4 * N ^ 2 * (C * EvenC H k 0 beta hb u)) / d :=
      div_le_div_of_nonneg_right hnum hd.le
    _ = _ := by ring

end
