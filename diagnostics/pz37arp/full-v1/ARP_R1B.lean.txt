private theorem p4Q2PZ37ARP_tensor_sum
    {α : Type u} {γ : Type v} [MeasurableSpace α] [MeasurableSpace γ]
    {μ : Measure α} {ν : Measure γ} [SFinite μ] [SFinite ν]
    (z : Lp ℝ 2 μ ⊗[ℝ] Lp ℝ 2 ν) :
    ∃ (d : ℕ) (a : Fin d → Lp ℝ 2 μ) (b : Fin d → Lp ℝ 2 ν),
      realL2ExternalTensorLift z = ∑ j, realL2ExternalTensor (a j) (b j) := by
  classical
  induction z using TensorProduct.induction_on with
  | zero =>
      exact ⟨0, Fin.elim0, Fin.elim0, by simp⟩
  | tmul a b =>
      exact ⟨1, fun _ => a, fun _ => b, by simp⟩
  | add x y hx hy =>
      obtain ⟨d,a,b,hd⟩ := hx
      obtain ⟨e,c,f,he⟩ := hy
      refine ⟨d+e, Fin.append a c, Fin.append b f, ?_⟩
      calc
        realL2ExternalTensorLift (x+y) = realL2ExternalTensorLift x + realL2ExternalTensorLift y := map_add _ _ _
        _ = (∑ j, realL2ExternalTensor (a j) (b j)) + ∑ j, realL2ExternalTensor (c j) (f j) := by rw [hd,he]
        _ = _ := by rw [Fin.sum_univ_add]; simp only [Fin.append_left, Fin.append_right]
