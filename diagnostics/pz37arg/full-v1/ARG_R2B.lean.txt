private theorem p4Q2PZ37ARG_eventual_aux (a q r : ℝ)
    (hq : 0 ≤ q) (hqr : q < r) (hr : 0 < r) :
    ∀ᶠ n : ℕ in atTop, q^n*a ≤ r^(n+1) := by
  have hlim : Tendsto (fun n : ℕ => (q/r)^n) atTop (𝓝 0) :=
    tendsto_pow_atTop_nhds_zero_of_lt_one (div_nonneg hq hr.le)
      ((div_lt_one hr).2 hqr)
  have hconst : Tendsto (fun _ : ℕ => a) atTop (𝓝 a) :=
    tendsto_const_nhds
  have ht : Tendsto (fun n : ℕ => (q/r)^n*a) atTop (𝓝 0) := by
    simpa only [zero_mul] using hlim.mul hconst
  have hev := ht (Iio_mem_nhds hr)
  change ∀ᶠ n : ℕ in atTop, (q/r)^n*a < r at hev
  filter_upwards [hev] with n hn
  have hrn : 0 < r^n := pow_pos hr _
  have heq : (q/r)^n*a = (q^n*a)/r^n := by
    rw [div_pow]
    ring
  have hh : (q^n*a)/r^n < r := heq ▸ hn
  have hh' := (div_lt_iff₀ hrn).1 hh
  calc
    q^n*a ≤ r*r^n := hh'.le
    _ = r^(n+1) := (pow_succ' r n).symm
