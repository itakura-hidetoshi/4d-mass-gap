/-- The unchanged APW observable depends only on the four original links. -/
theorem p4Q2PZ37AQN_fourSupport_dependsOn
    (H : ℕ) (hH : 0 < H) (c0 c1 : ℝ) :
    Supported H (↑(S4 H hH) : Set (PeriodicHypercubicEvenEdge H))
      (fun A => W H hH c0 c1 ((P H).positiveRestriction A)) := by
  classical
  intro A B hAB
  have he : (fun i : Fin 4 => Slice H hH ((P H).positiveRestriction A)
      (p4Q2PZ37AMS_primarySpatialSliceEdgeEmbedding H i)) =
    (fun i : Fin 4 => Slice H hH ((P H).positiveRestriction B)
      (p4Q2PZ37AMS_primarySpatialSliceEdgeEmbedding H i)) := by
    funext i
    have hm : Edge4 H hH i ∈ S4 H hH := by
      unfold p4Q2PZ37AQN_fourSupport
      exact Finset.mem_image.mpr ⟨i, Finset.mem_univ i, rfl⟩
    exact hAB (Edge4 H hH i) hm
  have hh : Hol H 2 (Slice H hH ((P H).positiveRestriction A)) =
      Hol H 2 (Slice H hH ((P H).positiveRestriction B)) := by
    unfold p4Q2PZ37AMS_primarySpatialSliceCyclicHolonomy
    exact congrArg haarFinFourCyclicPlaquetteWord he
  change c0 * Mode 0 (Hol H 2 (Slice H hH ((P H).positiveRestriction A))) +
      c1 * Mode 1 (Hol H 2 (Slice H hH ((P H).positiveRestriction A))) =
    c0 * Mode 0 (Hol H 2 (Slice H hH ((P H).positiveRestriction B))) +
      c1 * Mode 1 (Hol H 2 (Slice H hH ((P H).positiveRestriction B)))
  rw [hh]
