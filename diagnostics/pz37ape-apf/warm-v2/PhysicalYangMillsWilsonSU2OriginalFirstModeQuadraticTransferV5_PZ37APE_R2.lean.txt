/-- Transport a pointwise relative bound to the actual L2 norm. No
probability normalization or finite-dimensional hypothesis is needed. -/
private theorem p4Q2PZ37APE_relativeError_L2
    {α : Type*} [MeasurableSpace α] {mu : Measure α}
    (f g : Lp ℝ 2 mu) (a b : ℝ) (hb : 0 ≤ b)
    (h : ∀ᵐ x ∂mu, |f x + a * g x| ≤ b * g x) :
    ‖f + a • g‖ ≤ b * ‖g‖ := by
  have hAdd := Lp.coeFn_add f (a • g)
  have hA := Lp.coeFn_smul a g
  have hB := Lp.coeFn_smul b g
  have hle : ∀ᵐ x ∂mu, ‖(f + a • g) x‖ ≤ ‖(b • g) x‖ := by
    filter_upwards [hAdd, hA, hB, h] with x hx hax hbx hErr
    change (f + a • g) x = f x + (a • g) x at hx
    change (a • g) x = a * g x at hax
    change (b • g) x = b * g x at hbx
    calc
      ‖(f + a • g) x‖ = |f x + a * g x| := by
        rw [hx, hax, Real.norm_eq_abs]
      _ ≤ b * g x := hErr
      _ ≤ b * |g x| := mul_le_mul_of_nonneg_left (le_abs_self (g x)) hb
      _ = ‖(b • g) x‖ := by
        rw [hbx, Real.norm_eq_abs, abs_mul, abs_of_nonneg hb]
  calc
    _ ≤ ‖b • g‖ := Lp.norm_le_norm_of_ae_le hle
    _ = b * ‖g‖ := by rw [norm_smul, Real.norm_eq_abs, abs_of_nonneg hb]

norm relative to the exact image of its constant unit vector. -/

    (H : ℕ) (beta : ℝ) (hbeta : 0 ≤ beta) (hsmall : 16 * beta ≤ 1) :
    ‖Tr H 2 (by norm_num) beta hbeta (V H 1 + (beta / 4) • C H 2)‖ ≤
      195 * beta^2 * ‖Tr H 2 (by norm_num) beta hbeta (C H 2)‖ := by
  let mu := Mu H 2
  let t := Tr H 2 (by norm_num) beta hbeta
  let v : P H 2 := V H 1
  let c : P H 2 := C H 2
  let f : Lp ℝ 2 mu := ((t v : P H 2) : Lp ℝ 2 mu)
  let g : Lp ℝ 2 mu := ((t c : P H 2) : Lp ℝ 2 mu)
  have hf : f =ᵐ[mu] Rep H 2 beta (v : Lp ℝ 2 mu) :=
    p4Q2PZ37APE_transfer_ae H beta hbeta v
  have hg : g =ᵐ[mu] Rep H 2 beta (c : Lp ℝ 2 mu) :=
    p4Q2PZ37APE_transfer_ae H beta hbeta c
  have hErr : ∀ᵐ A ∂mu, |f A + beta / 4 * g A| ≤ 195 * beta^2 * g A := by
    filter_upwards [hf, hg] with A hA hB
    rw [hA, hB]
    exact p4Q2PZ37APD_originalPhysicalFirstMode_relativeTransferError_le H beta hbeta hsmall A
  have h := p4Q2PZ37APE_relativeError_L2 f g (beta / 4) (195 * beta^2)

  change ‖t (v + (beta / 4) • c)‖ ≤ 195 * beta^2 * ‖t c‖
  rw [map_add, map_smul]
  exact h

uniformly in H, with no lower bound on the raw operator norm assumed. -/
