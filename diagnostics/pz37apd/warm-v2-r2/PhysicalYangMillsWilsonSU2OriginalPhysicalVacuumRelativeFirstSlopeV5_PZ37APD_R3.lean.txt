/-- An a.e. relative error survives testing against any nonnegative
real L2 vector. This is a genuine L2 inner product, not point sampling. -/
private theorem p4Q2PZ37APD_relativeError_inner
    {α : Type*} [MeasurableSpace α] {mu : Measure α}
    (w f g : Lp ℝ 2 mu) (a b : ℝ)
    (hw : ∀ᵐ x ∂mu, 0 ≤ w x)
    (h : ∀ᵐ x ∂mu, |f x + a * g x| ≤ b * g x) :
    |inner ℝ w f + a * inner ℝ w g| ≤ b * inner ℝ w g := by
  have hf : Integrable (fun x => w x * f x) mu := by
    simpa only [periodicHypercubicEven_real_inner_eq_mul] using
      (MeasureTheory.L2.integrable_inner (𝕜 := ℝ) w f)
  have hg : Integrable (fun x => w x * g x) mu := by
    simpa only [periodicHypercubicEven_real_inner_eq_mul] using
      (MeasureTheory.L2.integrable_inner (𝕜 := ℝ) w g)
  have hLo : ∀ᵐ x ∂mu, -(b * (w x * g x)) ≤
      w x * f x + a * (w x * g x) := by
    filter_upwards [hw, h] with x hx hErr
    have hh := mul_le_mul_of_nonneg_left (abs_le.mp hErr).1 hx
    nlinarith only [hh]
  have hHi : ∀ᵐ x ∂mu, w x * f x + a * (w x * g x) ≤
      b * (w x * g x) := by
    filter_upwards [hw, h] with x hx hErr
    have hh := mul_le_mul_of_nonneg_left (abs_le.mp hErr).2 hx
    nlinarith only [hh]
  have hLoInt := integral_mono_ae ((hg.const_mul b).neg)
    (hf.add (hg.const_mul a)) hLo
  have hHiInt := integral_mono_ae (hf.add (hg.const_mul a))
    (hg.const_mul b) hHi
  rw [integral_neg', integral_add' hf (hg.const_mul a),
    integral_const_mul, integral_const_mul] at hLoInt
  rw [integral_add' hf (hg.const_mul a),
    integral_const_mul, integral_const_mul] at hHiInt
  simp only [MeasureTheory.L2.inner_def, periodicHypercubicEven_real_inner_eq_mul]
  exact abs_le.mpr ⟨hLoInt, hHiInt⟩

    (H : ℕ) (beta : ℝ) (hbeta : 0 ≤ beta) (hsmall : 16 * beta ≤ 1) :
    |inner ℝ (Vac H 2 (by norm_num) beta hbeta) (V H 1) +

  let t := Tr H 2 (by norm_num) beta hbeta
  let omega : P H 2 := Vac H 2 (by norm_num) beta hbeta
  let v : P H 2 := V H 1
  let c : P H 2 := C H 2
  let w : Lp ℝ 2 mu := (omega : Lp ℝ 2 mu)
  let f : Lp ℝ 2 mu := ((t v : P H 2) : Lp ℝ 2 mu)
  let g : Lp ℝ 2 mu := ((t c : P H 2) : Lp ℝ 2 mu)
  have hw : ∀ᵐ A ∂mu, 0 ≤ w A :=
    periodicHypercubicEvenSpecialUnitaryPhysicalOneSlabNonnegativeTopEigenvector_ae_nonnegative
      H 2 (by norm_num) beta hbeta
  have hf : f =ᵐ[mu] Rep H 2 beta (v : Lp ℝ 2 mu) :=
    p4Q2PZ37APD_transfer_ae H beta hbeta v
  have hg : g =ᵐ[mu] Rep H 2 beta (c : Lp ℝ 2 mu) :=
    p4Q2PZ37APD_transfer_ae H beta hbeta c
  have hErr : ∀ᵐ A ∂mu, |f A + beta / 4 * g A| ≤ 195 * beta^2 * g A := by
    filter_upwards [hf, hg] with A hA hB
    rw [hA, hB]
    exact p4Q2PZ37APD_originalPhysicalFirstMode_relativeTransferError_le H beta hbeta hsmall A
  have h := p4Q2PZ37APD_relativeError_inner w f g (beta / 4) (195 * beta^2) hw hErr
  change |inner ℝ omega (t v) + beta / 4 * inner ℝ omega (t c)| ≤
    195 * beta^2 * inner ℝ omega (t c) at h
  have hEigen : t omega = ‖t‖ • omega :=
    periodicHypercubicEvenSpecialUnitaryPhysicalOneSlabNonnegativeTopEigenvector_eigen
      H 2 (by norm_num) beta hbeta
  have hMove (u : P H 2) : inner ℝ omega (t u) = ‖t‖ * inner ℝ omega u := by
    calc
      _ = inner ℝ (t omega) u :=
        (periodicHypercubicEvenSpecialUnitaryPhysicalOneSlabTransferOperator_isSymmetric
          H 2 (by norm_num) beta hbeta omega u).symm
      _ = _ := by
        rw [hEigen]
        exact real_inner_smul_left omega u ‖t‖
  rw [hMove v, hMove c] at h
  have hPos : 0 < ‖t‖ :=
    periodicHypercubicEvenSpecialUnitaryPhysicalOneSlabTransferOperator_norm_pos
      H 2 (by norm_num) beta hbeta
  have hEq : ‖t‖ * inner ℝ omega v + beta / 4 * (‖t‖ * inner ℝ omega c) =
      ‖t‖ * (inner ℝ omega v + beta / 4 * inner ℝ omega c) := by ring
  rw [hEq, abs_mul, abs_of_pos hPos] at h
  change |inner ℝ omega v + beta / 4 * inner ℝ omega c| ≤
    195 * beta^2 * inner ℝ omega c
  apply (mul_le_mul_iff_of_pos_left hPos).mp
  calc
    _ ≤ 195 * beta^2 * (‖t‖ * inner ℝ omega c) := h
    _ = _ := by ring

end
