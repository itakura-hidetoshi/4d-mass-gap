theorem p4Q2PZ37AQA_shifted_support_residual_budget
    (H k l : ℕ) (s : Set (PeriodicHypercubicEvenEdge H))
    (hp : ∀ e ∈ s, periodicHypercubicEvenEdgeSide H e = ReflectionEdgeSide.positive)
    (hkl : ∀ e ∈ s, (e.1 0).val + k + l ≤ H) :
    ∀ e ∈ Shift H (k : ℤ) '' s,
      periodicHypercubicEvenEdgeSide H e = ReflectionEdgeSide.positive ∧
      (e.1 0).val + l ≤ H := by
  rintro e' ⟨e, he, rfl⟩
  have hb := hkl e he
  have hk : (e.1 0).val + k ≤ H := by omega
  have hn : (e.1 0).val + k < L H := by
    simp only [PeriodicHypercubicEvenSideLength]
    omega
  have hv : ((Shift H (k : ℤ) e).1 0).val = (e.1 0).val + k :=
    periodicHypercubicEvenIntegerTemporalEdgeTranslation_sourceTime_val_of_add_lt H k e hn
  exact ⟨p4Q2PZ37APZ_edgeShift_positive_of_budget H k e (hp e he) hk, by omega⟩

/-- A genuine intermediate half restriction and re-section do not change the
two-step reading. The combined budget also certifies the direct reading. -/
theorem p4Q2PZ37AQA_localRead_comp
    (H k l : ℕ) (s : Set (PeriodicHypercubicEvenEdge H))
    (f : (PeriodicHypercubicEvenEdge H → G) → ℝ)
    (hf : Supported H s f)
    (hp : ∀ e ∈ s, periodicHypercubicEvenEdgeSide H e = ReflectionEdgeSide.positive)
    (hkl : ∀ e ∈ s, (e.1 0).val + k + l ≤ H)
    (x : O H 2) :
    Read H k f (Step H 2 (-(l : ℤ)) x) = Read H (k + l) f x := by
  have hk : ∀ e ∈ s, (e.1 0).val + k ≤ H := by
    intro e he
    have hh := hkl e he
    omega
  have hsum : -((k + l : ℕ) : ℤ) = -(k : ℤ) + -(l : ℤ) := by omega
  have he := p4Q2PZ37AQA_localRead_full_eq H k s f hf hp hk
    (Tau (L H) (-(l : ℤ)) (Section H 2 x))
  have he' : Tau (L H) (-((k + l : ℕ) : ℤ)) (Section H 2 x) =
      Tau (L H) (-(k : ℤ)) (Tau (L H) (-(l : ℤ)) (Section H 2 x)) := by
    rw [hsum]
    exact periodicHypercubicIntegerTemporalConfigurationTranslation_add_apply
      (L H) (-(k : ℤ)) (-(l : ℤ)) (Section H 2 x)
  exact he.trans (congrArg f he').symm

law. No all-step positive algebra covariance is used. -/

    Read H (k + l) (fun A => W H hH c0 c1 ((P H).positiveRestriction A)) x := by
  apply p4Q2PZ37AQA_localRead_comp H k l
    (p4Q2PZ37APZ_timeOneEdgeSupport H hH)
    (fun A => W H hH c0 c1 ((P H).positiveRestriction A))
    (p4Q2PZ37APZ_timeOneFullObservable_dependsOn H hH c0 c1)
  · rintro e ⟨a, rfl⟩
    exact periodicHypercubicEvenPositiveHalfInteriorSpatialEdge_positive H (⟨0, hH⟩, a)
  · rintro e ⟨a, rfl⟩
    change ((periodicHypercubicEvenPositiveHalfInteriorSpatialEdge H
      (⟨0, hH⟩, a)).1 0).val + k + l ≤ H

end
