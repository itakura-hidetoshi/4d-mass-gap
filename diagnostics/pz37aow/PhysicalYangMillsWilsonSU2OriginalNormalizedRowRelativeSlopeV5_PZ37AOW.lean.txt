/-- Strict positivity of the original finite-volume row mass, from its
already certified pointwise global minorization floor. No uniform floor
in H is claimed; this is only the justification for division. -/
private theorem p4Q2PZ37AOW_originalRowMass_pos
    (H : ℕ) (beta : ℝ) (hbeta : 0 ≤ beta)
    (A : PeriodicHypercubicEvenSpecialUnitarySpatialSliceConfiguration H 2) :
    0 < (∫ B,
      periodicHypercubicEvenSpecialUnitaryTemporalGaugeOneSlabKernel H 2 beta A B
      ∂(periodicHypercubicEvenSpecialUnitarySpatialSliceHaarMeasure H 2)) := by
  let Cfg := PeriodicHypercubicEvenSpecialUnitarySpatialSliceConfiguration H 2
  let mu := periodicHypercubicEvenSpecialUnitarySpatialSliceHaarMeasure H 2
  let kernel : Cfg → ℝ := fun B =>
    periodicHypercubicEvenSpecialUnitaryTemporalGaugeOneSlabKernel H 2 beta A B
  let floor : ℝ :=
    periodicHypercubicEvenSpecialUnitaryPhysicalOneSlabGlobalMinorizationFloor H beta
  haveI : IsProbabilityMeasure mu := by dsimp [mu]; infer_instance
  have hKernelCont : Continuous kernel :=
    p4Q2PZ37AOW_fixedLeftKernel_continuous H beta A
  have hKernelInt : Integrable kernel mu :=
    hKernelCont.integrable_of_hasCompactSupport (HasCompactSupport.of_compactSpace _)
  have hPoint (B : Cfg) : floor ≤ kernel B :=
    periodicHypercubicEvenSpecialUnitaryPhysicalOneSlabGlobalMinorizationFloor_le_kernel
      H 2 (by norm_num) beta hbeta A B
  have hFloorInt : (∫ _ : Cfg, floor ∂mu) = floor := by
    exact integral_eq_const (Filter.Eventually.of_forall fun _ => rfl)
  have hZfloor : floor ≤ (∫ B : Cfg, kernel B ∂mu) := by
    calc
      floor = (∫ _ : Cfg, floor ∂mu) := hFloorInt.symm
      _ ≤ (∫ B : Cfg, kernel B ∂mu) := by
        apply integral_mono_ae (integrable_const floor) hKernelInt
        filter_upwards [] with B
        exact hPoint B
  change 0 < (∫ B : Cfg, kernel B ∂mu)
  exact lt_of_lt_of_le
    (periodicHypercubicEvenSpecialUnitaryPhysicalOneSlabGlobalMinorizationFloor_pos H beta)
    hZfloor

/-- The relative error after normalizing by the actual row mass and positive
coupling is bounded by 32*beta, with no global action-budget factor.
The weighted flipped coefficient itself is not replaced by -1/4. -/
theorem p4Q2PZ37AOW_originalFirstRowHaarMoment_normalizedRelativeSlopeError_le
    (H : ℕ) (beta : ℝ) (hbeta : 0 < beta) (hsmall : 8 * beta ≤ 1)
    (A : PeriodicHypercubicEvenSpecialUnitarySpatialSliceConfiguration H 2) :
    |(∫ B,
      periodicHypercubicEvenSpecialUnitaryTemporalGaugeOneSlabKernel H 2 beta A B *
        Chi (p4Q2PZ37AMS_primarySpatialSliceCyclicHolonomy H 2 B)
      ∂(periodicHypercubicEvenSpecialUnitarySpatialSliceHaarMeasure H 2)) /
        (beta * (∫ B,
          periodicHypercubicEvenSpecialUnitaryTemporalGaugeOneSlabKernel H 2 beta A B
          ∂(periodicHypercubicEvenSpecialUnitarySpatialSliceHaarMeasure H 2))) +
      (∫ B,
        p4Q2PZ37AOH_originalRightCenterTwistActionGap H A B *
        periodicHypercubicEvenSpecialUnitaryTemporalGaugeOneSlabKernel H 2 beta A
          (p4Q2PZ37ANV_singleLinkCenterTwist H
            (p4Q2PZ37AMS_primarySpatialSliceEdge H 0) B) *
        Chi (p4Q2PZ37AMS_primarySpatialSliceCyclicHolonomy H 2 B)
        ∂(periodicHypercubicEvenSpecialUnitarySpatialSliceHaarMeasure H 2)) /
        (2 * (∫ B,
          periodicHypercubicEvenSpecialUnitaryTemporalGaugeOneSlabKernel H 2 beta A B
          ∂(periodicHypercubicEvenSpecialUnitarySpatialSliceHaarMeasure H 2)))| ≤
      32 * beta := by
  let I : ℝ := ∫ B,
    periodicHypercubicEvenSpecialUnitaryTemporalGaugeOneSlabKernel H 2 beta A B *
      Chi (p4Q2PZ37AMS_primarySpatialSliceCyclicHolonomy H 2 B)
    ∂(periodicHypercubicEvenSpecialUnitarySpatialSliceHaarMeasure H 2)
  let D : ℝ := ∫ B,
    p4Q2PZ37AOH_originalRightCenterTwistActionGap H A B *
      periodicHypercubicEvenSpecialUnitaryTemporalGaugeOneSlabKernel H 2 beta A
        (p4Q2PZ37ANV_singleLinkCenterTwist H
          (p4Q2PZ37AMS_primarySpatialSliceEdge H 0) B) *
      Chi (p4Q2PZ37AMS_primarySpatialSliceCyclicHolonomy H 2 B)
    ∂(periodicHypercubicEvenSpecialUnitarySpatialSliceHaarMeasure H 2)
  let Z : ℝ := ∫ B,
    periodicHypercubicEvenSpecialUnitaryTemporalGaugeOneSlabKernel H 2 beta A B
    ∂(periodicHypercubicEvenSpecialUnitarySpatialSliceHaarMeasure H 2)
  have hZ : 0 < Z := p4Q2PZ37AOW_originalRowMass_pos H beta hbeta.le A
  have hResp : |(-2 : ℝ) * I - beta * D| ≤ 64 * beta^2 * Z :=
    p4Q2PZ37AOW_originalFixedLeftRowLocalLinearResponse_abs_le_rowMass
      H beta hbeta.le hsmall A
  have hScale : (-2 : ℝ) * (I + beta * D / 2) = (-2 : ℝ) * I - beta * D := by ring
  rw [← hScale, abs_mul] at hResp
  norm_num at hResp
  have hEq : I / (beta * Z) + D / (2 * Z) = (I + beta * D / 2) / (beta * Z) := by
    field_simp [ne_of_gt hbeta, ne_of_gt hZ] <;> ring
  change |I / (beta * Z) + D / (2 * Z)| ≤ 32 * beta
  rw [hEq, abs_div, abs_of_pos (mul_pos hbeta hZ)]
  apply (div_le_iff₀ (mul_pos hbeta hZ)).2
  calc
    |I + beta * D / 2| ≤ (64 * beta^2 * Z) / 2 := by linarith only [hResp]
    _ = (32 * beta) * (beta * Z) := by ring

end
