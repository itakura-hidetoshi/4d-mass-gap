private theorem p4Q2PZ37ARM_trunc_aux {ι : Type i} (rho : ι → ℝ)
    (g : ℝ → ℝ≥0∞) (d : ℝ) (hcap : ∀ k, rho k ≤ d) :
    (∫⁻ t in Ioi (0:ℝ), N rho t * g t) =
      ∫⁻ t in Ioc (0:ℝ) d, N rho t * g t := by
  let f : ℝ → ℝ≥0∞ := fun t => N rho t * g t
  have hs := lintegral_inter_add_diff (μ := volume) f (Ioi (0:ℝ))
    (measurableSet_Ioc : MeasurableSet (Ioc (0:ℝ) d))
  have hi : Ioi (0:ℝ) ∩ Ioc 0 d = Ioc 0 d := by
    ext t
    simp only [mem_inter_iff, mem_Ioi, mem_Ioc]
    tauto
  have hz : (∫⁻ t in Ioi (0:ℝ) \ Ioc 0 d, f t) = 0 := by
    apply setLIntegral_eq_zero (measurableSet_Ioi.diff measurableSet_Ioc)
    intro t ht
    have htd : d < t := by
      by_contra! h
      exact ht.2 ⟨ht.1,h⟩
    have he : {k : ι | t ≤ rho k} = ∅ := by
      apply Set.eq_empty_of_forall_notMem
      intro k hk
      exact (not_le_of_gt htd) (hk.trans (hcap k))
    simp [f, p4Q2PZ37ARL_spectralCount, he]
  rw [hi, hz, add_zero] at hs
  exact hs.symm

private theorem p4Q2PZ37ARM_integral_aux (d a q : ℝ)
    (hd : 0 ≤ d) (ha : 0 ≤ a) (hq : -1 < q) :
    (∫⁻ t in Ioc (0:ℝ) d, ENNReal.ofReal (a * t^q)) =
      ENNReal.ofReal (a * d^(q+1)/(q+1)) := by
  have hi : IntervalIntegrable (fun t : ℝ => t^q) volume 0 d :=
    intervalIntegral.intervalIntegrable_rpow' hq
  have hm : IntegrableOn (fun t : ℝ => a*t^q) (Ioc 0 d) :=
    (hi.const_mul a).1
  have hnon : 0 ≤ᵐ[volume.restrict (Ioc 0 d)] (fun t : ℝ => a*t^q) := by
    filter_upwards [self_mem_ae_restrict (measurableSet_Ioc : MeasurableSet (Ioc (0:ℝ) d))]
      with t ht
    exact mul_nonneg ha (Real.rpow_nonneg ht.1.le _)
  rw [← ofReal_integral_eq_lintegral_ofReal hm hnon,
    ← intervalIntegral.integral_of_le hd, intervalIntegral.integral_const_mul,
    _root_.integral_rpow (Or.inl hq), Real.zero_rpow (by linarith : q+1 ≠ 0), sub_zero]
  rw [mul_div_assoc]

private theorem p4Q2PZ37ARM_weight_aux {ι : Type i} (rho : ι → ℝ)
    (d A w p : ℝ) (m : ℕ) (hA : 0 ≤ A) (hw : 0 ≤ w)
    (henv : ∀ t ∈ Ioc (0:ℝ) d, N rho t ≤ ENNReal.ofReal (A*t^(-p))) :
    (∫⁻ t in Ioc (0:ℝ) d, N rho t * ENNReal.ofReal (w*t^m)) ≤
      ∫⁻ t in Ioc (0:ℝ) d, ENNReal.ofReal ((w*A)*t^((m:ℝ)-p)) := by
  apply lintegral_mono_ae
  filter_upwards [self_mem_ae_restrict (measurableSet_Ioc : MeasurableSet (Ioc (0:ℝ) d))]
    with t ht
  have ht0 : 0 < t := ht.1
  have hp : t^(-p)*t^(m:ℝ) = t^((m:ℝ)-p) := by
    rw [← Real.rpow_add ht0, show -p+(m:ℝ) = (m:ℝ)-p by ring]
  calc
    _ ≤ ENNReal.ofReal (A*t^(-p)) * ENNReal.ofReal (w*t^m) :=
      mul_le_mul_right' (henv t ht) _
    _ = ENNReal.ofReal ((A*t^(-p))*(w*t^m)) :=
      (ENNReal.ofReal_mul (mul_nonneg hA (Real.rpow_nonneg ht0.le _))).symm
    _ = ENNReal.ofReal ((w*A)*t^((m:ℝ)-p)) := by
      congr 1
      rw [← Real.rpow_natCast t m]
      calc
        _ = (w*A)*(t^(-p)*t^(m:ℝ)) := by ring
        _ = _ := by rw [hp]

theorem p4Q2PZ37ARM_moment_layercake_cap
    (H : ℕ) (beta : ℝ) (hb : 0 ≤ beta) (n : ℕ)
    (s : ℝ) (hs : 8 < s) (hcut : beta ≤ Cut s hs)
    {ι : Type i} (b : HilbertBasis ι ℝ (K H 2 h2 beta hb))
    (rho : ι → ℝ) (he : ∀ k, Step H 2 h2 beta hb (b k) = rho k • b k) :
    ENNReal.ofReal (M H beta hb n) =
      ∫⁻ t in Ioc (0:ℝ) c, N rho t * ENNReal.ofReal (2*((n:ℝ)+1)*t^(2*n+1)) := by
  have hcap (k : ι) : rho k ≤ c :=

/-- A real-exponent power envelope gives an explicit finite moment bound.
The strict exponent condition is required for integrability at zero. -/
theorem p4Q2PZ37ARM_moment_le_powerEnvelope
    (H : ℕ) (beta : ℝ) (hb : 0 ≤ beta) (n : ℕ)
    (s : ℝ) (hs : 8 < s) (hcut : beta ≤ Cut s hs)
    {ι : Type i} (b : HilbertBasis ι ℝ (K H 2 h2 beta hb))
    (rho : ι → ℝ) (he : ∀ k, Step H 2 h2 beta hb (b k) = rho k • b k)
    (A p : ℝ) (hA : 0 ≤ A) (hp : p < 2*((n:ℝ)+1))
    (henv : ∀ t ∈ Ioc (0:ℝ) c, N rho t ≤ ENNReal.ofReal (A*t^(-p))) :
    M H beta hb n ≤ (2*((n:ℝ)+1)*A)/(2*((n:ℝ)+1)-p) *
      c^(2*((n:ℝ)+1)-p) := by
  have hw : 0 ≤ 2*((n:ℝ)+1) := by positivity
  have hqeq : ((2*n+1 : ℕ) : ℝ)-p+1 = 2*((n:ℝ)+1)-p := by
    push_cast
    ring
  have hq : -1 < ((2*n+1 : ℕ) : ℝ)-p := by linarith only [hqeq,hp]
  have hd : 0 ≤ c := by norm_num
  have hh := p4Q2PZ37ARM_weight_aux rho c A (2*((n:ℝ)+1)) p (2*n+1) hA hw henv
  rw [← p4Q2PZ37ARM_moment_layercake_cap H beta hb n s hs hcut b rho he] at hh
  rw [p4Q2PZ37ARM_integral_aux c _ _ hd (mul_nonneg hw hA) hq, hqeq] at hh
  have hden : 0 < 2*((n:ℝ)+1)-p := sub_pos.mpr hp
  have hnon : 0 ≤ (2*((n:ℝ)+1)*A)/(2*((n:ℝ)+1)-p) * c^(2*((n:ℝ)+1)-p) :=
    mul_nonneg (div_nonneg (mul_nonneg hw hA) hden.le) (Real.rpow_nonneg hd _)
  apply (ENNReal.ofReal_le_ofReal_iff hnon).mp
  refine hh.trans_eq ?_
  congr 1
  rw [mul_div_assoc, mul_comm (c^(2*((n:ℝ)+1)-p))]

      0 ≤ A ∧ p < 2*((H:ℝ)+1) ∧
