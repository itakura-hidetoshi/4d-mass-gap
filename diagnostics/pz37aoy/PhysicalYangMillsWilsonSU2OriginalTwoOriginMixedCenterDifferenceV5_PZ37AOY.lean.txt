Diagnostic candidate. On the original finite-volume SU(2) spatial slice,
the two native primary origin links are distinct and no other plaquette
contains both (ANV, including H=0). Every crossing term uses only one
right-boundary link. Consequently the mixed center difference cancels every
term except the primary spatial plaquette. Its half-weighted contribution
is exactly -2 times the original primary Haar character.

This is a pointwise identity of the unchanged original action, not an
independence assumption on plaquettes, a physical spectral gap, or a
continuum mass-gap proof. It does not import either pending AOW or AOX.

/-- The full original spatial action's mixed difference isolates exactly
the original primary plaquette, with its original spatial weight one. -/
theorem p4Q2PZ37AOY_originalSpatialAction_twoOriginMixedDifference
    (H : ℕ) (B : Cfg H 2) :
    W H 2 B - W H 2 (T H (E H 0) B) -
      W H 2 (T H (E H 3) B) +
      W H 2 (T H (E H 3) (T H (E H 0) B)) =
      -4 * Chi (Cyc H 2 B) := by
  classical
  let Plaq := PeriodicHypercubicEvenSpatialSlicePlaquette H
  let F : Plaq → Cfg H 2 → ℝ := fun p B =>
    specialUnitaryWilsonPlaquetteEnergy 2 (Hol B p)
  let B0 := T H (E H 0) B
  let B3 := T H (E H 3) B
  let B03 := T H (E H 3) B0
  let p0 := P0 H
  have hComm : B03 = T H (E H 0) B3 :=
    p4Q2PZ37AOY_originTwists_commute H B
  have hNon (p : Plaq) (hp : p ≠ p0) :
      F p B - F p B0 - F p B3 + F p B03 = 0 := by
    by_cases h0 : E H 0 ∈ pz37zaSpatialPlaquetteSupport H p
    · have h3 : E H 3 ∉ pz37zaSpatialPlaquetteSupport H p := by
        intro he
        exact hp (p4Q2PZ37ANV_primaryOriginTwoLinks_unique H p h0 he)
      have ha : F p B3 = F p B :=
        p4Q2PZ37ANV_singleLinkCenterTwist_energy_of_not_mem H p (E H 3) h3 B
      have hb : F p B03 = F p B0 :=
        p4Q2PZ37ANV_singleLinkCenterTwist_energy_of_not_mem H p (E H 3) h3 B0
      linarith only [ha, hb]
    · have ha : F p B0 = F p B :=
        p4Q2PZ37ANV_singleLinkCenterTwist_energy_of_not_mem H p (E H 0) h0 B
      have hb : F p B03 = F p B3 := by
        rw [hComm]
        exact p4Q2PZ37ANV_singleLinkCenterTwist_energy_of_not_mem H p (E H 0) h0 B3
      linarith only [ha, hb]
  have hPrimary (B' : Cfg H 2) :
      F p0 B' = 1 - Chi (Cyc H 2 B') := by
    change specialUnitaryWilsonPlaquetteEnergy 2 (Hol B' (P0 H)) =
      1 - Chi (Cyc H 2 B')
    rw [p4Q2PZ37ANV_originalPrimarySpatialEnergy_eq_cyclic H B']
    dsimp [p4Q2PZ35C_originalOneLinkCharacter]
    ring
  have hc0 : Chi (Cyc H 2 B0) = -Chi (Cyc H 2 B) :=
    p4Q2PZ37AOY_originTwist_character_odd H (E H 0) (Or.inl rfl) B
  have hc3 : Chi (Cyc H 2 B3) = -Chi (Cyc H 2 B) :=
    p4Q2PZ37AOY_originTwist_character_odd H (E H 3) (Or.inr rfl) B
  have hc03 : Chi (Cyc H 2 B03) = Chi (Cyc H 2 B) := by
    change Chi (Cyc H 2 (T H (E H 3) B0)) = _
    rw [p4Q2PZ37AOY_originTwist_character_odd H (E H 3) (Or.inr rfl) B0,
      hc0, neg_neg]
  have hOne : F p0 B - F p0 B0 - F p0 B3 + F p0 B03 =
      -4 * Chi (Cyc H 2 B) := by
    rw [hPrimary B, hPrimary B0, hPrimary B3, hPrimary B03, hc0, hc3, hc03]
    ring
  change W H 2 B - W H 2 B0 - W H 2 B3 + W H 2 B03 = _
  calc
    _ = (∑ p : Plaq, F p B - F p B0 - F p B3 + F p B03) := by
      rw [periodicHypercubicEvenSpecialUnitarySpatialSliceWilsonAction_eq_finset_sum H 2 B,
        periodicHypercubicEvenSpecialUnitarySpatialSliceWilsonAction_eq_finset_sum H 2 B0,
        periodicHypercubicEvenSpecialUnitarySpatialSliceWilsonAction_eq_finset_sum H 2 B3,
        periodicHypercubicEvenSpecialUnitarySpatialSliceWilsonAction_eq_finset_sum H 2 B03]
      simp only [F, Finset.sum_sub_distrib, Finset.sum_add_distrib]
    _ = F p0 B - F p0 B0 - F p0 B3 + F p0 B03 := by
      apply Finset.sum_eq_single p0
      · intro p _ hp
        exact hNon p hp
      · intro hp
        exact (hp (Finset.mem_univ p0)).elim
    _ = _ := hOne

/-- The original temporal crossing action has zero mixed center difference:
each crossing summand contains only a single native right link. -/
theorem p4Q2PZ37AOY_originalCrossingAction_twoOriginMixedDifference_zero
    (H : ℕ) (A B : Cfg H 2) :
    X H 2 A B - X H 2 A (T H (E H 0) B) -
      X H 2 A (T H (E H 3) B) +
      X H 2 A (T H (E H 3) (T H (E H 0) B)) = 0 := by
  classical
  let Link := PeriodicHypercubicEvenSpatialSliceLink H
  let F : Link → Cfg H 2 → ℝ := fun e B =>
    specialUnitaryWilsonPlaquetteEnergy 2 ((A e)⁻¹ * B e)
  let B0 := T H (E H 0) B
  let B3 := T H (E H 3) B
  let B03 := T H (E H 3) B0
  have h03 : E H 0 ≠ E H 3 := p4Q2PZ37AOJ_primaryOriginLinks_ne H
  have hComm : B03 = T H (E H 0) B3 :=
    p4Q2PZ37AOY_originTwists_commute H B
  have hEach (e : Link) :
      F e B - F e B0 - F e B3 + F e B03 = 0 := by
    by_cases he : e = E H 0
    · have hne : e ≠ E H 3 := fun h => h03 (he.symm.trans h)
      have ha : B3 e = B e := by
        simp [B3, p4Q2PZ37ANV_singleLinkCenterTwist, hne]
      have hb : B03 e = B0 e := by
        simp [B03, p4Q2PZ37ANV_singleLinkCenterTwist, hne]
      have hFa : F e B3 = F e B := by
        dsimp only [F]
        rw [ha]
      have hFb : F e B03 = F e B0 := by
        dsimp only [F]
        rw [hb]
      linarith only [hFa, hFb]
    · have ha : B0 e = B e := by
        simp [B0, p4Q2PZ37ANV_singleLinkCenterTwist, he]
      have hb : B03 e = B3 e := by
        rw [hComm]
        simp [p4Q2PZ37ANV_singleLinkCenterTwist, he]
      have hFa : F e B0 = F e B := by
        dsimp only [F]
        rw [ha]
      have hFb : F e B03 = F e B3 := by
        dsimp only [F]
        rw [hb]
      linarith only [hFa, hFb]
  change X H 2 A B - X H 2 A B0 - X H 2 A B3 + X H 2 A B03 = 0
  calc
    _ = (∑ e : Link, F e B - F e B0 - F e B3 + F e B03) := by
      rw [periodicHypercubicEvenSpecialUnitaryTemporalGaugeCrossingAction_eq_finset_sum H 2 A B,
        periodicHypercubicEvenSpecialUnitaryTemporalGaugeCrossingAction_eq_finset_sum H 2 A B0,
        periodicHypercubicEvenSpecialUnitaryTemporalGaugeCrossingAction_eq_finset_sum H 2 A B3,
        periodicHypercubicEvenSpecialUnitaryTemporalGaugeCrossingAction_eq_finset_sum H 2 A B03]
      simp only [F, Finset.sum_sub_distrib, Finset.sum_add_distrib]
    _ = 0 := by
      apply Finset.sum_eq_zero
      intro e _
      exact hEach e
