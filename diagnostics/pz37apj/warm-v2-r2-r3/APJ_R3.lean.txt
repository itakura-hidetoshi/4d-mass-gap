private theorem p4Q2PZ37APJ_norm_ge_one_generic
    {E : Type*} [NormedAddCommGroup E] [InnerProductSpace ℝ E]
    (v c : E) (q : ℝ) (hv : ‖v‖ = 1)
    (hcv : inner ℝ c v = 0) : 1 ≤ ‖v + q • c‖ := by
  have hi : inner ℝ v (q • c) = 0 := by
    calc
      _ = inner ℝ (q • c) v := real_inner_comm (q • c) v
      _ = q * inner ℝ c v := real_inner_smul_left c v q
      _ = 0 := by rw [hcv, mul_zero]
  have hsq : ‖v + q • c‖ ^ 2 = ‖v‖ ^ 2 + ‖q • c‖ ^ 2 := by
    simpa only [pow_two] using norm_add_sq_eq_norm_sq_add_norm_sq_real hi
  rw [hv] at hsq
  apply le_of_sq_le_sq _ (norm_nonneg _)
  nlinarith only [hsq, sq_nonneg ‖q • c‖]

  let omega : P H 2 := Vac H 2 (by norm_num) beta hbeta
  let v : P H 2 := V H 1
  let c : P H 2 := C H 2

    H beta hbeta hsmall

    rw [sub_mul, hqa]
    ring

    calc
      |q - beta / 4| * a = |(q - beta / 4) * a| := by
        rw [abs_mul, abs_of_pos ha]

      v + q • c = v + ((beta / 4 + (q - beta / 4)) • c) := by rw [hcoef]
