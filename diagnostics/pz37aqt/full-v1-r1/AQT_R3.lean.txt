/-- Finite Gram PSD descends to the actual signed product-Haar quadratic
through a continuous RKHS feature. -/
theorem p4Q2PZ37AQT_psd_pair_integral_nonneg
    {Y : Type} [TopologicalSpace Y] [CompactSpace Y] [SecondCountableTopology Y]
    [MeasurableSpace Y] [BorelSpace Y]
    (mu : Measure Y) [IsFiniteMeasure mu]
    {k : Y → Y → ℝ} (hk : Continuous (fun p : Y × Y => k p.1 p.2))
    (cert : PSD Y k) (f : Y → ℝ) (hf : Continuous f) :
    0 ≤ ∫ p : Y × Y, f p.1 * k p.1 p.2 * f p.2 ∂(mu.prod mu) := by
  let F : RealHilbertKernelFeature Y k := cert.toHilbertFeature
  have hF : Continuous F.feature := F.continuous_feature_of_continuous_kernel hk
  let g : Y → F.FeatureHilbert := fun x => f x • F.feature x
  have hg : Continuous g := hf.smul hF
  have hi : Integrable g mu :=
    hg.integrable_of_hasCompactSupport (HasCompactSupport.of_compactSpace _)
  have hp : Integrable (fun p : Y × Y => inner ℝ (g p.1) (g p.2)) (mu.prod mu) :=
    ((hg.comp continuous_fst).inner (hg.comp continuous_snd)).integrable_of_hasCompactSupport
      (HasCompactSupport.of_compactSpace _)
  have he : (∫ p : Y × Y, f p.1 * k p.1 p.2 * f p.2 ∂(mu.prod mu)) =
      ∫ p : Y × Y, inner ℝ (g p.1) (g p.2) ∂(mu.prod mu) := by
    apply integral_congr_ae
    filter_upwards with p
    rw [F.kernel_eq_inner p.1 p.2]
    simp only [g, real_inner_smul_left, real_inner_smul_right]
    ring
  rw [he, MeasureTheory.integral_prod _ hp]
  change 0 ≤ ∫ x, ∫ y, inner ℝ (f x • F.feature x) (f y • F.feature y) ∂mu ∂mu
  rw [F.weighted_inner_doubleIntegral_eq_norm_integral_sq mu f hi]
  exact sq_nonneg _
