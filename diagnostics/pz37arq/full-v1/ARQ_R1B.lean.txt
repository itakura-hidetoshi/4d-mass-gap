private theorem p4Q2PZ37ARQ_residual_aux
    {E : Type u} [NormedAddCommGroup E] [InnerProductSpace ℝ E]
    {ι : Type i} (w : ι → E) (hw : Orthonormal ℝ w)
    (z : E) (rho : ι → ℝ) (hc : ∀ k, inner ℝ (w k) z = rho k)
    (S : Finset ι) :
    ‖z - ∑ k ∈ S, rho k • w k‖^2 = ‖z‖^2 - ∑ k ∈ S, (rho k)^2 := by
  let v : E := ∑ k ∈ S, rho k • w k
  have hv : ‖v‖^2 = ∑ k ∈ S, (rho k)^2 := by
    rw [← real_inner_self_eq_norm_sq]
    simpa [v, pow_two] using hw.inner_sum rho rho S
  have hi : inner ℝ v z = ∑ k ∈ S, (rho k)^2 := by
    dsimp only [v]
    rw [sum_inner]
    apply Finset.sum_congr rfl
    intro k _
    rw [real_inner_smul_left, hc k, pow_two]
  change ‖z-v‖^2 = _
  have hswap : inner ℝ z v = inner ℝ v z := real_inner_comm v z
  rw [norm_sub_sq_real, hswap, hi, hv]
  ring

    (H : ℕ) (beta : ℝ) (hb : 0 ≤ beta)
    {ι : Type i} (B : HilbertBasis ι ℝ (K H 2 h2 beta hb)) :
    Orthonormal ℝ (p4Q2PZ37ARQ_diagTensor H beta hb B) := by
  classical
  rw [orthonormal_iff_ite]
  intro j k
  change inner ℝ
    (realL2ExternalTensor (Emb H 2 h2 beta hb (B j)) (Emb H 2 h2 beta hb (B j)))
    (realL2ExternalTensor (Emb H 2 h2 beta hb (B k)) (Emb H 2 h2 beta hb (B k))) = _
  rw [realL2ExternalTensor_inner]
  have hh := (Emb H 2 h2 beta hb).inner_map_map (B j) (B k)
  rw [hh, (orthonormal_iff_ite.mp B.orthonormal) j k]
  by_cases h : j = k <;> simp [h]

theorem p4Q2PZ37ARQ_truncation_separable
    (H : ℕ) (beta : ℝ) (hb : 0 ≤ beta)
    {ι : Type i} (B : HilbertBasis ι ℝ (K H 2 h2 beta hb))
    (rho : ι → ℝ) (S : Finset ι) :
    ∃ (a b : Fin S.card → Lp ℝ 2 (Mu H 2)),
      (show Lp ℝ 2 ((Mu H 2).prod (Mu H 2)) from
        p4Q2PZ37ARQ_truncation H beta hb B rho S) =
        ∑ j, realL2ExternalTensor (a j) (b j) := by
  classical
  let e : Fin S.card ≃ S := S.equivFin.symm
  let f : ι → Lp ℝ 2 (Mu H 2) := fun k => Emb H 2 h2 beta hb (B k)
  let a : Fin S.card → Lp ℝ 2 (Mu H 2) := fun j => rho (e j).val • f (e j).val
  let b : Fin S.card → Lp ℝ 2 (Mu H 2) := fun j => f (e j).val
  refine ⟨a,b,?_⟩
  let g : ι → Lp ℝ 2 ((Mu H 2).prod (Mu H 2)) := fun k =>
    rho k • realL2ExternalTensor (f k) (f k)
  change (∑ k ∈ S, g k) = ∑ j, realL2ExternalTensor (a j) (b j)
  calc
    _ = ∑ k : S, g k.val := (Finset.sum_coe_sort S g).symm
    _ = ∑ j : Fin S.card, g (e j).val :=
      Fintype.sum_equiv e.symm _ _ (fun k => by simp)
    _ = _ := by
      apply Finset.sum_congr rfl
      intro j _
      exact (realL2ExternalTensor_smul_left (rho (e j).val) (f (e j).val) (f (e j).val)).symm

theorem p4Q2PZ37ARQ_projectedMinTerms_le_card
    (H : ℕ) (beta : ℝ) (hb : 0 ≤ beta)
    {ι : Type i} (B : HilbertBasis ι ℝ (K H 2 h2 beta hb))
    (rho : ι → ℝ) (he : ∀ k, Step H 2 h2 beta hb (B k) = rho k • B k)
    (S : Finset ι) (r : ℝ) (hr : 0 < r)
    (htail : (∑' k, (rho k)^2) - ∑ k ∈ S, (rho k)^2 < r^2) :
    p4Q2PZ37ARQ_projectedMinTerms H beta hb r hr ≤ S.card := by
  classical
  have hsq := p4Q2PZ37ARQ_truncation_errorSq H beta hb B rho he S
  have hc : ‖PK H beta hb - p4Q2PZ37ARQ_truncation H beta hb B rho S‖ < r := by
    by_contra! hn
    have hh := mul_self_le_mul_self hr.le hn
    nlinarith only [hsq, htail, hh]
  obtain ⟨a,b,hab⟩ := p4Q2PZ37ARQ_truncation_separable H beta hb B rho S
  apply Nat.find_min' (p4Q2PZ37ARQ_projected_exists_separable H beta hb r hr)
  exact ⟨a,b,by rw [← hab]; exact hc⟩

theorem p4Q2PZ37ARQ_exists_spectral_tail
    (H : ℕ) (beta : ℝ) (hb : 0 ≤ beta)
    {ι : Type i} (B : HilbertBasis ι ℝ (K H 2 h2 beta hb))
    (rho : ι → ℝ) (he : ∀ k, Step H 2 h2 beta hb (B k) = rho k • B k)
    (r : ℝ) (hr : 0 < r) :
    ∃ S : Finset ι, (∑' k, (rho k)^2) - ∑ k ∈ S, (rho k)^2 < r^2 := by
  have hs := p4Q2PZ37ARC_quadratic_hasSum H beta hb B rho he
  have ht : Tendsto (fun S : Finset ι => ∑ k ∈ S, (rho k)^2)
      atTop (𝓝 (‖PK H beta hb‖^2)) := hs
  obtain ⟨S,hS⟩ := (Metric.tendsto_nhds.mp ht (r^2) (sq_pos_of_pos hr)).exists
  rw [Real.dist_eq] at hS
  refine ⟨S,?_⟩
  rw [hs.tsum_eq]
  linarith only [(abs_lt.mp hS).1]
