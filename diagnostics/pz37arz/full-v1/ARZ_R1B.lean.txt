/-- Finite-rank bound counts only the indices not certified orthogonal to
the actual embedded space. The original approximation error is unchanged. -/
theorem p4Q2PZ37ARZ_filtered_approx
    {X : Type u} [NormedAddCommGroup X] [InnerProductSpace ℝ X] [CompleteSpace X]
    {α : Type v} [MeasurableSpace α] {μ : Measure α} [SFinite μ]
    (J : X →ₗᵢ[ℝ] Lp ℝ 2 μ) (T : X →L[ℝ] X)
    (L : Lp ℝ 2 (μ.prod μ))
    (hpair : ∀ x y : X, inner ℝ (T x) y =
      inner ℝ (realL2ExternalTensor (J x) (J y)) L)
    (d : ℕ) (a b : Fin d → Lp ℝ 2 μ) (s : Finset (Fin d))
    (hzero : ∀ j ∉ s, ∀ x : X, inner ℝ (b j) (J x) = 0) :
    ∃ F : X →L[ℝ] X,
      Nonempty (Module.Finite ℝ (LinearMap.range F.toLinearMap)) ∧
      Module.finrank ℝ (LinearMap.range F.toLinearMap) ≤ s.card ∧
      ‖T-F‖ ≤ ‖L - ∑ j, realL2ExternalTensor (a j) (b j)‖ := by
  classical
  let P := J.toContinuousLinearMap.adjoint
  let W : Submodule ℝ X := Submodule.span ℝ (Set.range (fun j : Fin d => P (b j)))
  let V : Submodule ℝ X := Submodule.span ℝ (Set.range (fun j : s => P (b j.val)))
  haveI : Module.Finite ℝ V := Module.Finite.span_of_finite ℝ (Set.finite_range _)
  have hWV : W ≤ V := by
    apply Submodule.span_le.mpr
    rintro y ⟨j,rfl⟩
    by_cases hj : j ∈ s
    · exact Submodule.subset_span ⟨⟨j,hj⟩,rfl⟩
    · have hi : inner ℝ (P (b j)) (P (b j)) = 0 := by
        change inner ℝ (J.toContinuousLinearMap.adjoint (b j)) (P (b j)) = 0
        rw [ContinuousLinearMap.adjoint_inner_left]
        exact hzero j hj (P (b j))
      rw [real_inner_self_eq_norm_sq] at hi
      have hz : P (b j) = 0 := norm_eq_zero.mp (by nlinarith [norm_nonneg (P (b j))])
      rw [hz]
      exact V.zero_mem
  have hV : Module.finrank ℝ V ≤ s.card := by
    simpa only [V, Fintype.card_coe] using
      (finrank_range_le_card (R := ℝ) (fun j : s => P (b j.val)))
  obtain ⟨F,hf,_,hd,_,he⟩ := p4Q2PZ37ARY_projected_approx J T L hpair d a b
  have hWVdim : Module.finrank ℝ W ≤ Module.finrank ℝ V :=
    Submodule.finrank_mono hWV
  exact ⟨F,hf,hd.trans (hWVdim.trans hV),he⟩
